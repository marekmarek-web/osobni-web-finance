    <script>
        // DATA PLACEHOLDER
        const FUNDS_DATA = {
            equity: {
                global: { name: "Global Equity Fund", risk: 5, expReturn: 0.08 },
                us: { name: "US Large Cap", risk: 5, expReturn: 0.09 },
            },
            bond: {
                gov: { name: "Government Bonds", risk: 2, expReturn: 0.03 },
                corp: { name: "Corporate Bonds", risk: 3, expReturn: 0.045 }
            },
            commodity: {
                gold: { name: "Gold ETF", risk: 4, expReturn: 0.02 }
            }
        };

        // START: APP_LOGIC
        class AppState {
            constructor() {
                this.STORAGE_KEY = 'financial_plan_state';
                this.currentStep = 1;
                this.totalSteps = 6;
                
                this.data = {
                    client: { name: '', dob: '', age: '', email: '', hasPartner: false },
                    partner: { name: '', dob: '' },
                    children: [], 
                    cashflow: { income: {}, expense: {} }, 
                    assets: { active: {}, liability: {} },
                    goals: [],
                    strategy: {}
                };

                this.init();
            }

            init() {
                this.loadState();
            }

            loadState() {
                const saved = localStorage.getItem(this.STORAGE_KEY);
                if (saved) {
                    try {
                        const parsed = JSON.parse(saved);
                        this.data = { ...this.data, ...parsed.data };
                        // this.currentStep = parsed.currentStep || 1; // Volitelně: obnovit krok
                    } catch (e) { console.error("Load Error", e); }
                }
            }

            saveState() {
                const toSave = {
                    data: this.data,
                    currentStep: this.currentStep,
                    timestamp: new Date().toISOString()
                };
                localStorage.setItem(this.STORAGE_KEY, JSON.stringify(toSave));
            }

            updateField(category, field, value) {
                if (this.data[category]) {
                    this.data[category][field] = value;
                    this.saveState();
                }
            }
            
            updateNestedData(category, subCategory, key, value) {
                if (!this.data[category][subCategory]) this.data[category][subCategory] = {};
                this.data[category][subCategory][key] = parseFloat(value) || 0;
                this.saveState();
            }

            addChild() {
                this.data.children.push({ id: Date.now(), name: '', dob: '' });
                this.saveState();
                return this.data.children[this.data.children.length - 1];
            }

            updateChild(id, field, value) {
                const child = this.data.children.find(c => c.id === id);
                if (child) {
                    child[field] = value;
                    this.saveState();
                }
            }

            removeChild(id) {
                this.data.children = this.data.children.filter(c => c.id !== id);
                this.saveState();
            }

            nextStep() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    return true;
                }
                return false;
            }

            prevStep() {
                if (this.currentStep > 1) {
                    this.currentStep--;
                    return true;
                }
                return false;
            }
        }

        // START: UI_RENDER
        class UI {
            constructor(appState) {
                this.app = appState;
                this.stepsContainer = document.getElementById('stepper-container');
                this.btnNext = document.getElementById('btn-next');
                this.btnPrev = document.getElementById('btn-prev');
                this.warningMsg = document.getElementById('warning-msg');
                
                this.childrenContainer = document.getElementById('children-container');
                this.btnAddChild = document.getElementById('btn-add-child');
                this.partnerSection = document.getElementById('partner-section');
                this.partnerToggle = document.getElementById('has-partner');

                this.initHeaderScroll();
                this.initListeners();
                this.renderStepper();
                this.updateView();
            }

            initHeaderScroll() {
                const header = document.querySelector('header');
                window.addEventListener('scroll', () => {
                    if (window.scrollY > 20) {
                        header.classList.remove('glass-header-top');
                        header.classList.add('glass-header-scrolled');
                    } else {
                        header.classList.add('glass-header-top');
                        header.classList.remove('glass-header-scrolled');
                    }
                });
            }

            initListeners() {
                this.btnNext.addEventListener('click', () => this.handleNext());
                this.btnPrev.addEventListener('click', () => this.handlePrev());

                // Krok 1
                this.bindInput('client-name', 'client', 'name');
                this.bindInput('client-email', 'client', 'email');
                
                const dobInput = document.getElementById('client-dob');
                dobInput.addEventListener('change', (e) => {
                    this.app.updateField('client', 'dob', e.target.value);
                    this.calculateAge(e.target.value, 'client-age');
                });

                this.partnerToggle.addEventListener('change', (e) => {
                    this.app.updateField('client', 'hasPartner', e.target.checked);
                    this.togglePartner(e.target.checked);
                });
                this.bindInput('partner-name', 'partner', 'name');
                document.getElementById('partner-dob').addEventListener('change', (e) => {
                    this.app.updateField('partner', 'dob', e.target.value);
                });

                this.btnAddChild.addEventListener('click', () => {
                    const child = this.app.addChild();
                    this.renderChildRow(child);
                });

                // Krok 2 & 3: Auto-calc
                document.querySelectorAll('input[type="number"]').forEach(input => {
                    input.addEventListener('input', (e) => {
                        const type = e.target.dataset.type;
                        if(type) {
                            const parts = e.target.id.split('-');
                            if(parts.length > 1) {
                                const category = type === 'asset' || type === 'liability' ? 'assets' : 'cashflow';
                                const sub = type === 'asset' ? 'active' : (type === 'liability' ? 'liability' : type);
                                this.app.updateNestedData(category, sub, parts[1], e.target.value);
                                this.recalcSums();
                            }
                        }
                    });
                });
            }

            bindInput(elementId, category, field) {
                const el = document.getElementById(elementId);
                if(el) {
                    el.addEventListener('input', (e) => {
                        this.app.updateField(category, field, e.target.value);
                        this.checkSoftValidation();
                    });
                }
            }

            calculateAge(dobVal, outputId) {
                if(!dobVal) return;
                const dob = new Date(dobVal);
                const diff_ms = Date.now() - dob.getTime();
                const age_dt = new Date(diff_ms); 
                const age = Math.abs(age_dt.getUTCFullYear() - 1970);
                document.getElementById(outputId).value = age;
                this.app.updateField('client', 'age', age);
            }

            togglePartner(show) {
                if(show) {
                    this.partnerSection.classList.remove('hidden');
                } else {
                    this.partnerSection.classList.add('hidden');
                }
            }

            renderChildRow(child) {
                const div = document.createElement('div');
                div.className = "flex gap-4 items-center animate-fadeIn";
                div.innerHTML = `
                    <input type="text" placeholder="Jméno dítěte" class="form-input flex-1 px-4 py-2" value="${child.name}">
                    <input type="date" class="form-input w-40 px-4 py-2" value="${child.dob}">
                    <button class="text-red-400 hover:text-red-600"><i class="fas fa-trash"></i></button>
                `;
                
                const inputs = div.querySelectorAll('input');
                inputs[0].addEventListener('input', (e) => this.app.updateChild(child.id, 'name', e.target.value));
                inputs[1].addEventListener('change', (e) => this.app.updateChild(child.id, 'dob', e.target.value));
                div.querySelector('button').addEventListener('click', () => {
                    div.remove();
                    this.app.removeChild(child.id);
                });

                if(this.childrenContainer.children.length > 0 && this.childrenContainer.children[0].classList.contains('text-center')) {
                    this.childrenContainer.innerHTML = '';
                }
                this.childrenContainer.appendChild(div);
            }

            recalcSums() {
                const inc = this.app.data.cashflow.income;
                const exp = this.app.data.cashflow.expense;
                
                const totalInc = Object.values(inc).reduce((a, b) => a + b, 0);
                const totalExp = Object.values(exp).reduce((a, b) => a + b, 0);
                
                document.getElementById('total-income').innerText = totalInc.toLocaleString() + ' Kč';
                document.getElementById('total-expense').innerText = totalExp.toLocaleString() + ' Kč';
                
                const balance = totalInc - totalExp;
                const badge = document.getElementById('cashflow-badge');
                badge.innerText = `Bilance: ${balance.toLocaleString()} Kč`;
                badge.className = `px-4 py-2 rounded-lg text-sm font-bold ${balance >= 0 ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'}`;

                const assets = this.app.data.assets.active;
                const liabilities = this.app.data.assets.liability;

                const totalAssets = Object.values(assets).reduce((a, b) => a + b, 0);
                const totalLiabilities = Object.values(liabilities).reduce((a, b) => a + b, 0);
                const netWorth = totalAssets - totalLiabilities;

                if(document.getElementById('total-assets')) {
                    document.getElementById('total-assets').innerText = totalAssets.toLocaleString() + ' Kč';
                    document.getElementById('total-liabilities').innerText = totalLiabilities.toLocaleString() + ' Kč';
                    document.getElementById('net-worth').innerText = netWorth.toLocaleString() + ' Kč';
                }
            }

            renderStepper() {
                this.stepsContainer.innerHTML = '';
                const stepTitles = ["Klient", "Cashflow", "Majetek", "Cíle", "Strategie", "Shrnutí"];
                for (let i = 1; i <= this.app.totalSteps; i++) {
                    const stepDiv = document.createElement('div');
                    stepDiv.className = `stepper-item flex-1 flex flex-col items-center`;
                    stepDiv.dataset.step = i;
                    stepDiv.innerHTML = `
                        <div class="stepper-circle mb-2">${i}</div>
                        <span class="text-xs font-bold text-slate-400 uppercase tracking-wider hidden md:block">${stepTitles[i-1]}</span>
                    `;
                    this.stepsContainer.appendChild(stepDiv);
                }
            }

            updateView() {
                const current = this.app.currentStep;
                
                document.querySelectorAll('.step-content').forEach(el => el.classList.remove('active'));
                const activeStep = document.getElementById(`step-${current}`);
                if(activeStep) activeStep.classList.add('active');

                document.querySelectorAll('.stepper-item').forEach(el => {
                    const stepNum = parseInt(el.dataset.step);
                    const circle = el.querySelector('.stepper-circle');
                    el.classList.remove('active', 'completed');
                    
                    if (stepNum < current) {
                        el.classList.add('completed');
                        circle.innerHTML = '<i class="fas fa-check"></i>';
                    } else if (stepNum === current) {
                        el.classList.add('active');
                        circle.innerText = stepNum;
                    } else {
                        circle.innerText = stepNum;
                    }
                });

                this.btnPrev.classList.toggle('hidden', current === 1);
                
                if (current === this.app.totalSteps) {
                    this.btnNext.innerHTML = '<i class="fas fa-print mr-2"></i> Exportovat PDF';
                    this.btnNext.classList.replace('btn-primary', 'bg-slate-900');
                    this.btnNext.classList.add('text-white');
                    document.getElementById('summary-client-name').innerText = this.app.data.client.name || 'Nevyplněno';
                    this.recalcSums();
                    document.getElementById('summary-net-worth').innerText = document.getElementById('net-worth').innerText;

                } else {
                    this.btnNext.innerHTML = 'Další <i class="fas fa-arrow-right ml-2"></i>';
                    this.btnNext.classList.remove('bg-slate-900', 'text-white');
                    this.btnNext.classList.add('btn-primary');
                }

                if(current === 1) {
                    document.getElementById('client-name').value = this.app.data.client.name;
                    document.getElementById('client-email').value = this.app.data.client.email || '';
                    document.getElementById('client-dob').value = this.app.data.client.dob || '';
                    document.getElementById('client-age').value = this.app.data.client.age || '';
                    document.getElementById('has-partner').checked = this.app.data.client.hasPartner;
                    this.togglePartner(this.app.data.client.hasPartner);
                    
                    this.childrenContainer.innerHTML = '';
                    if(this.app.data.children.length === 0) {
                         this.childrenContainer.innerHTML = '<div class="text-center py-6 text-slate-400 bg-slate-50 rounded-xl border border-dashed border-slate-200 text-sm">Žádné děti zatím nepřidány.</div>';
                    } else {
                        this.app.data.children.forEach(c => this.renderChildRow(c));
                    }
                }
                
                this.checkSoftValidation();
            }

            checkSoftValidation() {
                let missing = false;
                if(this.app.currentStep === 1 && !this.app.data.client.name) missing = true;
                
                if(missing) {
                    this.warningMsg.classList.remove('hidden');
                } else {
                    this.warningMsg.classList.add('hidden');
                }
            }

            handleNext() {
                if (this.app.currentStep === this.app.totalSteps) {
                    window.print();
                } else {
                    this.app.nextStep();
                    this.updateView();
                }
            }

            handlePrev() {
                this.app.prevStep();
                this.updateView();
            }
        }

        // INIT
        document.addEventListener('DOMContentLoaded', () => {
            const app = new AppState();
            const ui = new UI(app);
        });
    </script>
