            loadState() {
                const saved = localStorage.getItem(this.STORAGE_KEY);
                if (saved) {
                    try {
                        const parsed = JSON.parse(saved);
                        
                        // Deep Merge Logic
                        if (parsed && parsed.data) {
                            if(parsed.data.client) this.data.client = { ...this.data.client, ...parsed.data.client };
                            if(parsed.data.partner) this.data.partner = { ...this.data.partner, ...parsed.data.partner };
                            if(Array.isArray(parsed.data.children)) this.data.children = parsed.data.children;

                            if(parsed.data.cashflow) {
                                // ... existing cashflow merge ...
                                this.data.cashflow.incomeType = parsed.data.cashflow.incomeType || 'zamestnanec';
                                this.data.cashflow.incomeGross = parsed.data.cashflow.incomeGross || 0;
                                this.data.cashflow.reserveCash = parsed.data.cashflow.reserveCash || 0;
                                this.data.cashflow.reserveTargetMonths = parsed.data.cashflow.reserveTargetMonths || 6;
                                if(parsed.data.cashflow.incomes) this.data.cashflow.incomes = { ...this.data.cashflow.incomes, ...parsed.data.cashflow.incomes };
                                if(parsed.data.cashflow.expenses) this.data.cashflow.expenses = { ...this.data.cashflow.expenses, ...parsed.data.cashflow.expenses };
                            }
                            
                            if(parsed.data.assets) {
                                if(parsed.data.assets.active) this.data.assets.active = { ...this.data.assets.active, ...parsed.data.assets.active };
                                if(parsed.data.assets.liability) this.data.assets.liability = { ...this.data.assets.liability, ...parsed.data.assets.liability };
                                if(parsed.data.assets.cash !== undefined) this.data.assets = { ...this.data.assets, ...parsed.data.assets };
                            }
                            if(parsed.data.liabilities) {
                                this.data.liabilities = { ...this.data.liabilities, ...parsed.data.liabilities };
                            }

                            if(Array.isArray(parsed.data.goals)) {
                                this.data.goals = parsed.data.goals;
                            }

                            // Strategy Merge
                            if(parsed.data.strategy) {
                                this.data.strategy.profile = parsed.data.strategy.profile || 'balanced';
                                if(parsed.data.strategy.lumpsum) this.data.strategy.lumpsum = { ...this.data.strategy.lumpsum, ...parsed.data.strategy.lumpsum };
                                if(parsed.data.strategy.regular) this.data.strategy.regular = { ...this.data.strategy.regular, ...parsed.data.strategy.regular };
                                if(parsed.data.strategy.pension) this.data.strategy.pension = { ...this.data.strategy.pension, ...parsed.data.strategy.pension };
                            }
                        }
                    } catch (e) { 
                        console.error("Load Error (Legacy Data?)", e); 
                    }
                }
            }

            saveState() {
                const toSave = {

                    currentStep: this.currentStep,
                    timestamp: new Date().toISOString()
                };
                localStorage.setItem(this.STORAGE_KEY, JSON.stringify(toSave));
            }

                if (!this.data[category][subCategory]) this.data[category][subCategory] = {};
                this.data[category][subCategory][key] = parseFloat(value) || 0;
                this.saveState();
            }
            
            updateBalanceItem(category, key, value) {

                    this.saveState();
                }
            }

                if(!this.data.children) this.data.children = [];
                this.data.children.push({ id: Date.now(), name: '', birthDate: '' });
                this.saveState();
                return this.data.children[this.data.children.length - 1];
            }

            updateChild(id, field, value) {
                const child = this.data.children.find(c => c.id === id);
                if (child) {
                    child[field] = value;
                    this.saveState();
                }
            }

            removeChild(id) {
                this.data.children = this.data.children.filter(c => c.id !== id);
                this.saveState();
            }

            // Goals Management
            addGoal(rawGoal) {
                // rawGoal comes from UI: { type, name, amount (generic), horizon, strategy (number), initial, lumpsum }
                
                // 1. Determine Strategy Label
                let strategyLabel = "balanced";
                if(rawGoal.strategy >= 0.09) strategyLabel = "dynamic";
                else if(rawGoal.strategy <= 0.05) strategyLabel = "conservative";

                // 2. Determine Target Amount vs Income
                let targetMonthlyIncome = null;
                let targetAmount = null;
                let fvTarget = 0;

                if (rawGoal.type === 'renta') {
                    targetMonthlyIncome = rawGoal.amount;
                    // Renta calculation: (Monthly * 12) / 0.04 (4% rule)
                    fvTarget = (targetMonthlyIncome * 12) / 0.04;
                } else {
                    targetAmount = rawGoal.amount;
                    fvTarget = targetAmount;
                }

                // 3. Compute PMT & Net Needed
                const r = rawGoal.strategy / 12; // monthly rate
                const n = rawGoal.horizon * 12; // months
                
                const initialTotal = (rawGoal.initial || 0) + (rawGoal.lumpsum || 0);
                const fvExisting = initialTotal * Math.pow(1 + r, n);
                const netNeeded = Math.max(0, fvTarget - fvExisting);
                
                let pmt = 0;
                if (netNeeded > 0 && n > 0) {
                    pmt = netNeeded * r / (Math.pow(1 + r, n) - 1);
                }

                if(!this.data.goals) this.data.goals = [];
                
                const newGoal = {
                    id: Date.now(),
                    type: rawGoal.type,
                    name: rawGoal.name,
                    years: rawGoal.horizon,
                    strategy: strategyLabel,
                    annualRate: rawGoal.strategy,
                    targetMonthlyIncome: targetMonthlyIncome,
                    targetAmount: targetAmount,
                    initialAmount: rawGoal.initial || 0,
                    lumpSumNow: rawGoal.lumpsum || 0,
                    computed: {
                        fvTarget: fvTarget,
                        pmt: pmt,
                        netNeeded: netNeeded
                    }
                };

                this.data.goals.push(newGoal);
                this.saveState();
            }

            removeGoal(id) {
                this.data.goals = this.data.goals.filter(g => g.id !== id);
                this.saveState();
            }

            // Strategy Management
            updateStrategyProduct(product, field, value) {
                if(this.data.strategy[product]) {
                    this.data.strategy[product][field] = parseFloat(value) || 0;
                    this.saveState();
                }
            }

            setStrategyProfile(profile) {
                this.data.strategy.profile = profile;
                this.saveState();
            }

            nextStep() {

                }
                return false;
            }

            prevStep() {
                if (this.currentStep > 1) {
                    this.currentStep--;
                    return true;
                }
                return false;
            }
        }

            recalcStrategy() {
                const strat = this.app.data.strategy;
                let rate = 0.07;
                if(strat.profile === 'dynamic') rate = 0.09;
                if(strat.profile === 'conservative') rate = 0.05;

                // FV Helper: Lump Sum
                const calcFVLump = (amount, years) => {
                    return amount * Math.pow(1 + rate, years);
                };

                // FV Helper: Regular (Monthly)
                // FV = PMT * (((1 + r)^n - 1) / r)
                const calcFVReg = (monthly, years) => {
                    const r = rate / 12;
                    const n = years * 12;
                    if(r === 0) return monthly * n;
                    return monthly * ((Math.pow(1 + r, n) - 1) / r);
                };

                // 1. Lump Sum Product
                const ls = strat.lumpsum;
                const fvLs = calcFVLump(ls.amount, ls.years);
                document.getElementById('fv-lumpsum').innerText = Math.round(fvLs).toLocaleString('cs-CZ') + ' Kč';

                // 2. Regular Product
                const reg = strat.regular;
                const fvReg = calcFVReg(reg.amount, reg.years);
                document.getElementById('fv-regular').innerText = Math.round(fvReg).toLocaleString('cs-CZ') + ' Kč';

                // 3. Pension Product
                // (Simplified: treating as regular investment with same rate for projection)
                const pen = strat.pension;
                const fvPen = calcFVReg(pen.amount, pen.years);
                document.getElementById('fv-pension').innerText = Math.round(fvPen).toLocaleString('cs-CZ') + ' Kč';

                // Totals
                const totalFV = fvLs + fvReg + fvPen;
                const totalInvested = ls.amount + (reg.amount * reg.years * 12) + (pen.amount * pen.years * 12);
                const totalProfit = totalFV - totalInvested;

                document.getElementById('strat-total-fv').innerText = Math.round(totalFV).toLocaleString('cs-CZ') + ' Kč';
                document.getElementById('strat-total-invested').innerText = Math.round(totalInvested).toLocaleString('cs-CZ') + ' Kč';
                document.getElementById('strat-total-profit').innerText = Math.round(totalProfit).toLocaleString('cs-CZ') + ' Kč';
            }

            updateView() {
                const current = this.app.currentStep;
                
                document.querySelectorAll('.step-content').forEach(el => el.classList.remove('active'));
                const activeStep = document.getElementById(`step-${current}`);
                if(activeStep) activeStep.classList.add('active');

                document.querySelectorAll('.stepper-item').forEach(el => {
                    const stepNum = parseInt(el.dataset.step);
                    const circle = el.querySelector('.stepper-circle');
                    el.classList.remove('active', 'completed');
                    
                    if (stepNum < current) {
                        el.classList.add('completed');
                        circle.innerHTML = '<i class="fas fa-check"></i>';
                    } else if (stepNum === current) {
                        el.classList.add('active');
                        circle.innerText = stepNum;
                    } else {
                        circle.innerText = stepNum;
                    }
                });

                if(this.btnPrev) this.btnPrev.classList.toggle('hidden', current === 1);
                
                if (current === this.app.totalSteps) {
                    if(this.btnNext) {
                        this.btnNext.innerHTML = '<i class="fas fa-print mr-2"></i> Exportovat PDF';
                        this.btnNext.classList.replace('btn-primary', 'bg-slate-900');
                        this.btnNext.classList.add('text-white');
                    }
                    const summaryName = document.getElementById('summary-client-name');
                    if(summaryName) summaryName.innerText = this.app.data.client.name || 'Nevyplněno';
                    this.recalcSums();
                    const sumNW = document.getElementById('summary-net-worth');
                    const nw = document.getElementById('net-worth');
                    if(sumNW && nw) sumNW.innerText = nw.innerText;

                } else {
                    if(this.btnNext) {
                        this.btnNext.innerHTML = 'Další <i class="fas fa-arrow-right ml-2"></i>';
                        this.btnNext.classList.remove('bg-slate-900', 'text-white');
                        this.btnNext.classList.add('btn-primary');
                    }
                }

                if(current === 1) {
                    const elName = document.getElementById('client-name');
                    if(elName) elName.value = this.app.data.client.name;
                    
                    const elEmail = document.getElementById('client-email');
                    if(elEmail) elEmail.value = this.app.data.client.email || '';
                    
                    const elPhone = document.getElementById('client-phone');
                    if(elPhone) elPhone.value = this.app.data.client.phone || '';
                    
                    const elDob = document.getElementById('client-dob');
                    if(elDob) elDob.value = this.app.data.client.birthDate || '';
                    
                    const elAge = document.getElementById('client-age');
                    if(elAge) elAge.value = this.app.data.client.age || '--';
                    
                    const elPartnerCheck = document.getElementById('has-partner');
                    if(elPartnerCheck) {
                        elPartnerCheck.checked = this.app.data.client.hasPartner;
                        this.togglePartner(this.app.data.client.hasPartner);
                    }
                    
                    const elPName = document.getElementById('partner-name');
                    if(elPName) elPName.value = this.app.data.partner.name || '';
                    
                    const elPDob = document.getElementById('partner-dob');
                    if(elPDob) elPDob.value = this.app.data.partner.birthDate || '';

                    this.childrenContainer.innerHTML = '';
                    if(!this.app.data.children || this.app.data.children.length === 0) {
                         this.childrenContainer.innerHTML = '<div class="text-center py-6 text-slate-400 bg-white rounded-xl border border-dashed border-slate-200 text-sm italic">Žádné děti zatím nepřidány.</div>';
                    } else {
                        this.app.data.children.forEach(c => this.renderChildRow(c));
                    }
                }
                
                if(current === 2) {
                    const incomeType = document.getElementById('income-type');
                    if(incomeType) incomeType.value = this.app.data.cashflow.incomeType || 'zamestnanec';
                    
                    const incomeGross = document.getElementById('income-gross');
                    if(incomeGross) incomeGross.value = this.app.data.cashflow.incomeGross || '';

                    const reserve = document.getElementById('asset-cash-quick');

                    if(reserveMonths) {
                         const val = this.app.data.cashflow.reserveTargetMonths || 6;
                         reserveMonths.value = val;
                         document.getElementById('reserve-months-display').innerText = val;

                    this.recalcSums();

                        const el = document.getElementById(id);

                    restore('asset-realestate', a.realEstate);

                    restore('asset-pension', a.pension);
                    restore('asset-other', a.other);
                    
                    restore('liability-mortgage', l.mortgage);
                    restore('liability-loans', l.loans);
                    restore('liability-other', l.other);

                    this.recalcSums();

                    this.renderGoalsList();
                    setTimeout(() => this.renderGoalsChart(), 100); 
                }

                if(current === 5) {
                    // Initialize Strategy View
                    this.updateStrategyUI();
                    
                    // Restore inputs
                    const s = this.app.data.strategy;
                    const restoreStrat = (prod, data) => {
                        const elAmt = document.getElementById(`strat-${prod}-amount`);
                        const elYrs = document.getElementById(`strat-${prod}-years`);
                        if(elAmt) elAmt.value = data.amount || '';
                        if(elYrs) elYrs.value = data.years || '';
                    };
                    
                    restoreStrat('lumpsum', s.lumpsum);
                    restoreStrat('regular', s.regular);
                    restoreStrat('pension', s.pension);
                    
                    this.recalcStrategy();
                }

                this.checkInfoTip();
            }

            checkInfoTip() {
                let showWarning = false;
                const current = this.app.currentStep;

                if (current === 1) {
                     if (!this.app.data.client.name) showWarning = true;
                } else if (current === 2) {
                     const inc = this.app.data.cashflow.incomes || {};
                     const exp = this.app.data.cashflow.expenses || {};
                     const totalInc = Object.values(inc).reduce((a, b) => a + b, 0);
                     const totalExp = Object.values(exp).reduce((a, b) => a + b, 0);
                     if (totalInc === 0 && totalExp === 0) showWarning = true;
                }
                
                if(this.infoTip) {
                    if(showWarning) {
                        this.infoTip.innerHTML = '<i class="fas fa-exclamation-circle"></i> Tip: Některá pole nejsou vyplněna';
                        this.infoTip.classList.remove('hidden');
                    } else {
                        this.infoTip.classList.add('hidden');
                    }
                }
            }

            handleNext() {
                if (this.app.currentStep === this.app.totalSteps) {
                    window.print();
                } else {
                    this.app.nextStep();
                    this.updateView();
                }
            }

            handlePrev() {
                this.app.prevStep();
                this.updateView();
            }
        }
