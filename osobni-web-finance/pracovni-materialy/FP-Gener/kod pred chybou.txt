            init() {

                    // Pre-populate investments if empty

                            { id: 1, productKey: 'imperial', type: 'lump', amount: 0, years: 10, annualRate: 0.12, computed: {fv:0} },
                            { id: 2, productKey: 'creif', type: 'lump', amount: 0, years: 10, annualRate: 0.06, computed: {fv:0} },

                        ];
                        this.saveState();
                    }
                } catch(e) {
                    console.error("Critical State Error", e);
                }
            }

            loadState() {
                const saved = localStorage.getItem(this.STORAGE_KEY);
                if (saved) {
                    try {
                        const parsed = JSON.parse(saved);
                        
                        // Deep Merge Logic
                        if (parsed && parsed.data) {
                            if(parsed.data.client) this.data.client = { ...this.data.client, ...parsed.data.client };
                            if(parsed.data.partner) this.data.partner = { ...this.data.partner, ...parsed.data.partner };
                            if(Array.isArray(parsed.data.children)) this.data.children = parsed.data.children;

                            if(parsed.data.cashflow) {
                                // ... existing cashflow merge ...
                                this.data.cashflow.incomeType = parsed.data.cashflow.incomeType || 'zamestnanec';
                                this.data.cashflow.incomeGross = parsed.data.cashflow.incomeGross || 0;
                                this.data.cashflow.reserveCash = parsed.data.cashflow.reserveCash || 0;
                                this.data.cashflow.reserveTargetMonths = parsed.data.cashflow.reserveTargetMonths || 6;
                                if(parsed.data.cashflow.incomes) this.data.cashflow.incomes = { ...this.data.cashflow.incomes, ...parsed.data.cashflow.incomes };
                                if(parsed.data.cashflow.expenses) this.data.cashflow.expenses = { ...this.data.cashflow.expenses, ...parsed.data.cashflow.expenses };
                            }
                            
                            if(parsed.data.assets) {
                                if(parsed.data.assets.active) this.data.assets.active = { ...this.data.assets.active, ...parsed.data.assets.active };
                                if(parsed.data.assets.liability) this.data.assets.liability = { ...this.data.assets.liability, ...parsed.data.assets.liability };
                                if(parsed.data.assets.cash !== undefined) this.data.assets = { ...this.data.assets, ...parsed.data.assets };
                            }
                            if(parsed.data.liabilities) {
                                this.data.liabilities = { ...this.data.liabilities, ...parsed.data.liabilities };
                            }

                            if(Array.isArray(parsed.data.goals)) {
                                this.data.goals = parsed.data.goals.map(g => {
                                    // FORCE RECALC FOR RENTA TO FIX 6% LOGIC
                                    let fvTarget = g.amount;
                                    if (g.type === 'renta') {
                                        fvTarget = (g.amount * 12) / 0.06;
                                    } else {
                                        if (g.computed && g.computed.fvTarget) fvTarget = g.computed.fvTarget;
                                    }
                                    
                                    const r = (g.annualRate || 0.07) / 12;
                                    const n = (g.years || g.horizon || 1) * 12;
                                    const initialTotal = (g.initialAmount || 0) + (g.lumpSumNow || 0);
                                    const fvExisting = initialTotal * Math.pow(1 + r, n);
                                    const netNeeded = Math.max(0, fvTarget - fvExisting);
                                    
                                    let pmt = 0;
                                    if (netNeeded > 0 && n > 0) {
                                        pmt = netNeeded * r / (Math.pow(1 + r, n) - 1);
                                    }

                                    return {
                                        ...g,
                                        years: g.years || g.horizon,
                                        annualRate: g.annualRate || g.strategy,
                                        computed: { fvTarget, pmt, netNeeded }
                                    };
                                });
                            }

                            // Strategy Merge - Profile & Conservative
                            if(parsed.data.strategy) {
                                this.data.strategy.profile = parsed.data.strategy.profile || 'balanced';
                                this.data.strategy.conservativeMode = parsed.data.strategy.conservativeMode || false;
                            }
                            // Investments Merge
                            if(Array.isArray(parsed.data.investments)) {
                                this.data.investments = parsed.data.investments;
                            }
                        }
                    } catch (e) { 
                        console.error("Load Error (Legacy Data?)", e); 
                    }
                }
            }

            saveState() {
                const toSave = {
                    data: this.data,
                    currentStep: this.currentStep,
                    timestamp: new Date().toISOString()
                };
                localStorage.setItem(this.STORAGE_KEY, JSON.stringify(toSave));
            }

            updateField(category, field, value) {
                if (this.data[category]) {
                    this.data[category][field] = value;
                    this.saveState();
                }
            }
            
            updateNestedData(category, subCategory, key, value) {
                if (!this.data[category][subCategory]) this.data[category][subCategory] = {};
                this.data[category][subCategory][key] = parseFloat(value) || 0;
                this.saveState();
            }
            
            updateBalanceItem(category, key, value) {
                if (this.data[category]) {
                    this.data[category][key] = parseFloat(value) || 0;
                    this.saveState();
                }
            }

            addChild() {
                if(!this.data.children) this.data.children = [];
                this.data.children.push({ id: Date.now(), name: '', birthDate: '' });
                this.saveState();
                return this.data.children[this.data.children.length - 1];
            }

            updateChild(id, field, value) {
                const child = this.data.children.find(c => c.id === id);
                if (child) {
                    child[field] = value;
                    this.saveState();
                }
            }

            removeChild(id) {
                this.data.children = this.data.children.filter(c => c.id !== id);
                this.saveState();
            }

            // Goals Management
            addGoal(rawGoal) {
                // 1. Determine Strategy Label
                let strategyLabel = "balanced";
                if(rawGoal.strategy >= 0.09) strategyLabel = "dynamic";
                else if(rawGoal.strategy <= 0.05) strategyLabel = "conservative";

                // 2. Determine Target Amount vs Income
                let targetMonthlyIncome = null;
                let targetAmount = null;
                let fvTarget = 0;

                if (rawGoal.type === 'renta') {
                    targetMonthlyIncome = rawGoal.amount;
                    // Renta calculation: (Monthly * 12) / 0.06 (6% rule)
                    fvTarget = (targetMonthlyIncome * 12) / 0.06;
                } else {
                    targetAmount = rawGoal.amount;
                    fvTarget = targetAmount;
                }

                // 3. Compute PMT & Net Needed
                const r = rawGoal.strategy / 12; // monthly rate
                const n = rawGoal.horizon * 12; // months
                
                const initialTotal = (rawGoal.initial || 0) + (rawGoal.lumpsum || 0);
                const fvExisting = initialTotal * Math.pow(1 + r, n);
                const netNeeded = Math.max(0, fvTarget - fvExisting);
                
                let pmt = 0;
                if (netNeeded > 0 && n > 0) {
                    pmt = netNeeded * r / (Math.pow(1 + r, n) - 1);
                }

                if(!this.data.goals) this.data.goals = [];
                
                const newGoal = {
                    id: Date.now(),
                    type: rawGoal.type,
                    name: rawGoal.name,
                    years: rawGoal.horizon,
                    strategy: strategyLabel,
                    annualRate: rawGoal.strategy,
                    targetMonthlyIncome: targetMonthlyIncome,
                    targetAmount: targetAmount,
                    initialAmount: rawGoal.initial || 0,
                    lumpSumNow: rawGoal.lumpsum || 0,
                    computed: {
                        fvTarget: fvTarget,
                        pmt: pmt,
                        netNeeded: netNeeded
                    }
                };

                this.data.goals.push(newGoal);
                this.saveState();
            }

            removeGoal(id) {
                this.data.goals = this.data.goals.filter(g => g.id !== id);
                this.saveState();
            }

            // Strategy Management - Updated to array based
            updateInvestment(productKey, type, field, value) {
                const inv = this.data.investments.find(i => i.productKey === productKey && i.type === type);
                if(inv) {
                    inv[field] = parseFloat(value) || 0;
                    this.saveState();
                }
            }

            setStrategyProfile(profile) {
                this.data.strategy.profile = profile;
                this.saveState();
            }

            setConservativeMode(isConservative) {
                this.data.strategy.conservativeMode = isConservative;
                this.saveState();
            }

            nextStep() {
                if (this.currentStep < this.totalSteps) {
                    this.currentStep++;
                    return true;
                }
                return false;
            }

            prevStep() {
                if (this.currentStep > 1) {
                    this.currentStep--;
                    return true;
                }
                return false;
            }
        }

            calculateAge(dobStr, outputId) {
                if (!dobStr) return;
                const now = new Date();
                const currentYear = now.getFullYear();
                let age = '--';

                if (/^(19|20)\d{2}$/.test(dobStr.trim())) {
                    const year = parseInt(dobStr.trim(), 10);
                    if (year > 1900 && year <= currentYear) {
                        age = currentYear - year;
                    }
                } 
                else {
                    const parts = dobStr.match(/(\d{1,2})[\.\/](\d{1,2})[\.\/](\d{4})/);
                    if (parts) {
                        const day = parseInt(parts[1], 10);
                        const month = parseInt(parts[2], 10) - 1; 
                        const year = parseInt(parts[3], 10);
                        const birthDate = new Date(year, month, day);
                        if (birthDate.getFullYear() === year && birthDate.getMonth() === month) {
                            age = currentYear - year;
                            const m = now.getMonth() - month;
                            if (m < 0 || (m === 0 && now.getDate() < day)) {
                                age--;
                            }
                        }
                    }
                }

                const el = document.getElementById(outputId);
                if (el) el.value = age;
                if (age !== '--') this.app.updateField('client', 'age', age);
            }

            recalcSums() {
                try {
                    // Step 2 & 3 Sums (Same as before)
                    // ... (Code abbreviated for brevity, logic remains identical)
                    const inc = this.app.data.cashflow.incomes || {};
                    const exp = this.app.data.cashflow.expenses || {};
                    const totalInc = Object.values(inc).reduce((a, b) => a + b, 0);
                    const totalExp = Object.values(exp).reduce((a, b) => a + b, 0);
                    
                    const elInc = document.getElementById('total-income');
                    if(elInc) elInc.innerText = totalInc.toLocaleString('cs-CZ') + ' Kč';
                    const elExp = document.getElementById('total-expense');
                    if(elExp) elExp.innerText = totalExp.toLocaleString('cs-CZ') + ' Kč';
                    
                    const balance = totalInc - totalExp;
                    const badge = document.getElementById('cashflow-badge');
                    if(badge) {
                        if (balance > 0) {
                            badge.innerText = `Bilance: +${balance.toLocaleString('cs-CZ')} Kč`;
                            badge.className = `bg-green-100 text-green-700 px-4 py-2 rounded-lg text-sm font-bold transition-colors`;
                        } else {
                            badge.innerText = `Bilance: ${balance.toLocaleString('cs-CZ')} Kč`;
                            badge.className = balance < 0 ? `bg-red-100 text-red-700 px-4 py-2 rounded-lg text-sm font-bold transition-colors` : `bg-slate-100 text-slate-500 px-4 py-2 rounded-lg text-sm font-bold transition-colors`;
                        }
                    }
                    const surplusEl = document.getElementById('total-surplus');
                    if(surplusEl) {
                        surplusEl.innerText = balance.toLocaleString('cs-CZ') + ' Kč';
                        surplusEl.className = balance >= 0 ? 'text-2xl font-bold text-brand-main' : 'text-2xl font-bold text-red-500';
                    }

                    // Reserve logic
                    const reserveCash = this.app.data.cashflow.reserveCash || 0;
                    const reserveMonths = this.app.data.cashflow.reserveTargetMonths || 6;
                    const reserveTarget = totalExp * reserveMonths;
                    const reserveGap = Math.max(0, reserveTarget - reserveCash);
                    const targetEl = document.getElementById('reserve-target');
                    const gapEl = document.getElementById('reserve-gap');
                    if(targetEl) targetEl.innerText = `${reserveTarget.toLocaleString('cs-CZ')} Kč (${reserveMonths} měs.)`;
                    if(gapEl) gapEl.innerText = reserveGap > 0 ? `${reserveGap.toLocaleString('cs-CZ')} Kč` : "0 Kč";

                    // Assets
                    const assets = this.app.data.assets || {};
                    const liabilities = this.app.data.liabilities || {};
                    const totalAssets = Object.values(assets).reduce((a, b) => a + (parseFloat(b) || 0), 0);
                    const totalLiabilities = Object.values(liabilities).reduce((a, b) => a + (parseFloat(b) || 0), 0);
                    const netWorth = totalAssets - totalLiabilities;
                    
                    const elTotalAssets = document.getElementById('total-assets');
                    if(elTotalAssets) elTotalAssets.innerText = totalAssets.toLocaleString('cs-CZ') + ' Kč';
                    const elTotalLiabilities = document.getElementById('total-liabilities');
                    if(elTotalLiabilities) elTotalLiabilities.innerText = totalLiabilities.toLocaleString('cs-CZ') + ' Kč';
                    const elNetWorth = document.getElementById('net-worth');
                    if(elNetWorth) elNetWorth.innerText = netWorth.toLocaleString('cs-CZ') + ' Kč';

                    this.checkInfoTip();
                } catch(e) { console.error("Recalc Error", e); }
            }

            renderGoalsChart() {
                const ctx = document.getElementById('goals-chart');
                if(!ctx) return;

                if(this.goalsChart) {
                    this.goalsChart.destroy();
                    this.goalsChart = null;
                }

                if (!this.chartGoalId) {
                    this.goalsChart = new Chart(ctx, {
                        type: 'line',
                        data: {
                            labels: [],
                            datasets: []
                        },
                        options: {
                            responsive: true,
                            maintainAspectRatio: false,
                            plugins: {
                                title: {
                                    display: true,
                                    text: 'Vyberte cíl v seznamu pro zobrazení grafu',
                                    font: { size: 16, weight: 'normal' },
                                    color: '#94a3b8'
                                },
                                legend: { display: false }
                            },
                            scales: {
                                x: { display: false },
                                y: { display: false }
                            }
                        }
                    });
                    return;
                }

                const goal = this.app.data.goals.find(g => g.id === this.chartGoalId);
                if (!goal) return;

                const labels = Array.from({length: goal.years + 1}, (_, i) => `Rok ${i}`);
                
                // Safe computed access
                const computed = goal.computed || {};
                const target = computed.fvTarget || 0;
                const monthly = computed.pmt || 0;
                
                const targetData = new Array(goal.years + 1).fill(target);

                // Projection Curve based on stored computed values
                const r = goal.annualRate;
                const monthlyRate = r/12;
                const initialTotal = (goal.initialAmount || 0) + (goal.lumpSumNow || 0);

                const projectionData = labels.map((_, year) => {
                    const monthsPassed = year * 12;
                    const val_init = initialTotal * Math.pow(1 + monthlyRate, monthsPassed);
                    let val_monthly = 0;
                    if(monthlyRate > 0) {
                        val_monthly = monthly * (Math.pow(1 + monthlyRate, monthsPassed) - 1) / monthlyRate;
                    } else {
                        val_monthly = monthly * monthsPassed;
                    }
                    return val_init + val_monthly;
                });

                this.goalsChart = new Chart(ctx, {
                    type: 'line',
                    data: {
                        labels: labels,
                        datasets: [
                            {
                                label: 'Cílová částka',
                                data: targetData,
                                borderColor: '#ef4444', 
                                borderDash: [5, 5],
                                borderWidth: 2,
                                pointRadius: 0,
                                fill: false
                            },
                            {
                                label: 'Projekce portfolia',
                                data: projectionData,
                                borderColor: '#ea8d00', 
                                backgroundColor: 'rgba(234, 141, 0, 0.1)',
                                fill: true,
                                tension: 0.4,
                                borderWidth: 3
                            }
                        ]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        interaction: {
                            mode: 'index',
                            intersect: false,
                        },
                        plugins: {
                            legend: { 
                                display: true,
                                position: 'bottom'
                            },
                            tooltip: {
                                callbacks: {
                                    label: function(context) {
                                        let label = context.dataset.label || '';
                                        if (label) {
                                            label += ': ';
                                        }
                                        if (context.parsed.y !== null) {
                                            label += Math.round(context.parsed.y).toLocaleString('cs-CZ') + ' Kč';
                                        }
                                        return label;
                                    }
                                }
                            }
                        },
                        scales: {
                            y: {
                                beginAtZero: true,
                                ticks: {
                                    callback: function(value) {
                                        return (value / 1000000).toFixed(1) + ' M';
                                    }
                                },
                                grid: {
                                    color: '#f1f5f9'
                                }
                            },
                            x: {
                                grid: {
                                    display: false
                                }
                            }
                        }
                    }
                });
            }

            recalcStrategy() {
                let totalFV = 0;
                let totalInvested = 0;
                let totalLump = 0;
                let totalMonthly = 0;

                const invs = this.app.data.investments || [];
                const conservative = this.app.data.strategy.conservativeMode || false;
                
                // FV Helpers - adapted for specific rates
                const calcFVLump = (amount, years, rate) => amount * Math.pow(1 + rate, years);
                const calcFVReg = (monthly, years, rate) => {
                    const r = rate / 12;
                    const n = years * 12;
                    if(r === 0) return monthly * n;
                    return monthly * ((Math.pow(1 + r, n) - 1) / r);
                };

                invs.forEach(inv => {
                    let fv = 0;
                    // Apply conservative modifier (-2%) but floor at 0%
                    const annualRate = Math.max(0, (inv.annualRate || 0) - (conservative ? 0.02 : 0));

                    if(inv.type === 'lump') {
                        fv = calcFVLump(inv.amount || 0, inv.years || 0, annualRate);
                        totalInvested += (inv.amount || 0);
                        totalLump += (inv.amount || 0);
                    } else if(inv.type === 'monthly' || inv.type === 'pension') {
                        fv = calcFVReg(inv.amount || 0, inv.years || 0, annualRate);
                        totalInvested += ((inv.amount || 0) * (inv.years || 0) * 12);
                        totalMonthly += (inv.amount || 0);
                    }
                    totalFV += fv;
                    
                    // Store computed if needed
                    inv.computed = { fv: fv };
                });

                const totalProfit = totalFV - totalInvested;

                const elTotalFV = document.getElementById('strat-total-fv');
                if(elTotalFV) elTotalFV.innerText = Math.round(totalFV).toLocaleString('cs-CZ') + ' Kč';
                
                const elProfitLabel = document.getElementById('strat-profit-label');
                if(elProfitLabel) elProfitLabel.innerText = `+${Math.round(totalProfit).toLocaleString('cs-CZ')} zisk`;

                const elLump = document.getElementById('summary-lump-total');
                if(elLump) elLump.innerText = Math.round(totalLump).toLocaleString('cs-CZ') + ' Kč';
                
                const elMonthly = document.getElementById('summary-monthly-total');
                if(elMonthly) elMonthly.innerText = Math.round(totalMonthly).toLocaleString('cs-CZ') + ' Kč';
            }
